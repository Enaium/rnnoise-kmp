cmake_minimum_required(VERSION 3.16)

project(rnnoise C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Builds the JNI shared library by default. Native (cinterop) targets build
# only the static library by passing -DBUILD_JNI=OFF.
option(BUILD_JNI "Build the JNI shared library" ON)
option(RNN_ENABLE_X86_RTCD "Enable x86 runtime CPU dispatch (SSE4.1/AVX2)" ON)

set(RNNOISE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../rnnoise)

# ============================================================================
# Model weights
#
# The rnnoise main branch does not ship the trained default model: it is
# generated by `download_model.sh` (see the rnnoise repo README). We download
# the model tarball once and keep only the little-endian weights file, which
# is about 40% of the size of the big-endian variant and is the correct
# layout for every platform we target.
#
# The files are stored under c_api/ so the rnnoise submodule working tree is
# never modified.
# ============================================================================
set(C_API_DIR ${CMAKE_CURRENT_SOURCE_DIR}/c_api)
# The directory only contains build-time downloads (see below), so it is not
# tracked by git; make sure it exists before the download step.
file(MAKE_DIRECTORY ${C_API_DIR})
set(RNNOISE_DATA_C ${C_API_DIR}/rnnoise_data_little.c)
set(RNNOISE_DATA_H ${C_API_DIR}/rnnoise_data.h)

file(STRINGS ${RNNOISE_DIR}/model_version RNNOISE_MODEL_HASH)

if(NOT EXISTS ${RNNOISE_DATA_C} OR NOT EXISTS ${RNNOISE_DATA_H})
    message(STATUS "rnnoise model weights not found; downloading rnnoise_data-${RNNOISE_MODEL_HASH}.tar.gz ...")
    set(DOWNLOAD_SCRIPT ${CMAKE_CURRENT_BINARY_DIR}/download_model.sh)
    file(WRITE ${DOWNLOAD_SCRIPT}
"#!/bin/sh
set -e
cd \"$1\" || exit 1
hash=\"$2\"
mkdir -p .tmp-download
if command -v curl >/dev/null 2>&1; then
  curl -fsSL \"https://media.xiph.org/rnnoise/models/rnnoise_data-\${hash}.tar.gz\" -o .tmp-download/model.tar.gz
elif command -v wget >/dev/null 2>&1; then
  wget -q \"https://media.xiph.org/rnnoise/models/rnnoise_data-\${hash}.tar.gz\" -O .tmp-download/model.tar.gz
else
  echo \"Neither curl nor wget is available.\" >&2
  exit 1
fi
tar xzf .tmp-download/model.tar.gz -C .tmp-download src/rnnoise_data_little.c src/rnnoise_data.h
cp .tmp-download/src/rnnoise_data_little.c rnnoise_data_little.c
cp .tmp-download/src/rnnoise_data.h rnnoise_data.h
rm -rf .tmp-download
")
    execute_process(
        COMMAND sh ${DOWNLOAD_SCRIPT} ${C_API_DIR} ${RNNOISE_MODEL_HASH}
        RESULT_VARIABLE DOWNLOAD_RESULT
    )
    if(NOT DOWNLOAD_RESULT EQUAL 0)
        message(FATAL_ERROR "Failed to download rnnoise model weights from media.xiph.org")
    endif()
    message(STATUS "rnnoise model weights downloaded to ${C_API_DIR}/")
endif()

# ============================================================================
# rnnoise static library (submodule sources)
# ============================================================================
set(RNNOISE_SRC
    ${RNNOISE_DIR}/src/denoise.c
    ${RNNOISE_DIR}/src/rnn.c
    ${RNNOISE_DIR}/src/pitch.c
    ${RNNOISE_DIR}/src/kiss_fft.c
    ${RNNOISE_DIR}/src/celt_lpc.c
    ${RNNOISE_DIR}/src/nnet.c
    ${RNNOISE_DIR}/src/nnet_default.c
    ${RNNOISE_DIR}/src/parse_lpcnet_weights.c
    ${RNNOISE_DATA_C}
    ${RNNOISE_DIR}/src/rnnoise_tables.c
)

add_library(rnnoise STATIC ${RNNOISE_SRC})
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD)
target_include_directories(rnnoise PUBLIC
    ${RNNOISE_DIR}/include
    ${RNNOISE_DIR}/src
    ${C_API_DIR}
)

if(MSVC)
    target_compile_options(rnnoise PRIVATE /W3)
else()
    target_compile_options(rnnoise PRIVATE -Wall -fPIC -Wno-unused-function)
endif()

# Effective architecture: CMAKE_SYSTEM_PROCESSOR reflects the host even when
# cross-compiling via CMAKE_OSX_ARCHITECTURES, so prefer the explicit one on
# Apple.
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    set(EFFECTIVE_ARCH ${CMAKE_OSX_ARCHITECTURES})
else()
    set(EFFECTIVE_ARCH ${CMAKE_SYSTEM_PROCESSOR})
endif()

message(STATUS "EFFECTIVE_ARCH: ${EFFECTIVE_ARCH}")

if(EFFECTIVE_ARCH MATCHES "x86_64|AMD64|amd64|i686|x86|i386")
    if(RNN_ENABLE_X86_RTCD)
        target_compile_definitions(rnnoise PRIVATE RNN_ENABLE_X86_RTCD CPU_INFO_BY_C)
        target_sources(rnnoise PRIVATE
            ${RNNOISE_DIR}/src/x86/x86cpu.c
            ${RNNOISE_DIR}/src/x86/x86_dnn_map.c
        )
        set_source_files_properties(${RNNOISE_DIR}/src/x86/nnet_sse4_1.c PROPERTIES COMPILE_FLAGS "-msse4.1")
        if(MSVC)
            set_source_files_properties(${RNNOISE_DIR}/src/x86/nnet_avx2.c PROPERTIES COMPILE_FLAGS "/arch:AVX2")
        else()
            set_source_files_properties(${RNNOISE_DIR}/src/x86/nnet_avx2.c PROPERTIES COMPILE_FLAGS "-mavx -mfma -mavx2")
        endif()
        target_sources(rnnoise PRIVATE
            ${RNNOISE_DIR}/src/x86/nnet_sse4_1.c
            ${RNNOISE_DIR}/src/x86/nnet_avx2.c
        )
    endif()
endif()

# ============================================================================
# JNI shared library
# ============================================================================
if(BUILD_JNI)
    add_library(rnnoise_jni SHARED ${CMAKE_CURRENT_SOURCE_DIR}/jni_bridge.cpp)
    set_target_properties(rnnoise_jni PROPERTIES OUTPUT_NAME rnnoise_jni)
    if(WIN32)
        # MinGW prefixes DLLs with "lib"; strip it so the artifact is
        # rnnoise_jni.dll everywhere (NativeLoader expects no prefix).
        set_target_properties(rnnoise_jni PROPERTIES PREFIX "")
        target_compile_definitions(rnnoise_jni PRIVATE DLL_EXPORT)
    endif()

    if(ANDROID)
        find_library(log-lib log)
        if(log-lib)
            target_link_libraries(rnnoise_jni PRIVATE ${log-lib})
        endif()
    elseif(JNI_INCLUDE_DIR)
        target_include_directories(rnnoise_jni PRIVATE ${JNI_INCLUDE_DIR})
        if(JNI_INCLUDE_DIR_PLATFORM)
            target_include_directories(rnnoise_jni PRIVATE ${JNI_INCLUDE_DIR_PLATFORM})
        endif()
    else()
        find_package(JNI REQUIRED)
        target_include_directories(rnnoise_jni PRIVATE ${JNI_INCLUDE_DIRS})
    endif()

    target_link_libraries(rnnoise_jni PRIVATE rnnoise)
    target_link_libraries(rnnoise_jni PRIVATE m)
endif()
